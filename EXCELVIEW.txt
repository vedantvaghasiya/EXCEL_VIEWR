    <script type="module">

        // ---------------------------------------------------------------------
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'sheetvault-default';
        const firebaseConfig = JSON.parse(__firebase_config);
        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);

        // Strict Firestore paths under /artifacts/{appId}/public/data/...
        const ownerConfigDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'vault_config', 'owner_lock');
        const workbookDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'vault_workbooks', 'master_workbook');

        const LOCAL_OWNER_CACHE_KEY = `sheetvault_owner_${appId}`;
        const LOCAL_WORKBOOK_CACHE_KEY = `sheetvault_wb_${appId}`;
        const SESSION_UNLOCK_KEY = `sheetvault_unlocked_user_${appId}`;

        let firebaseUser = null;
        let currentUser = null; // { identifier: string, method: 'password' | 'google' }

        // ---------------------------------------------------------------------
        // SHA-256 PASSWORD HASHING (WEB CRYPTO API)
        // ---------------------------------------------------------------------
        async function hashPassword(password) {
            const encoder = new TextEncoder();
            const data = encoder.encode('sheetvault_salt_v1::' + password);
            const hashBuffer = await crypto.subtle.digest('SHA-256', data);
            const hashArray = Array.from(new Uint8Array(hashBuffer));
            return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
        }

        // ---------------------------------------------------------------------
        // SILENT BACKGROUND CLOUD AUTH + VAULT STATUS INITIALIZATION
        // ---------------------------------------------------------------------
        async function ensureFirebaseAuth() {
            if (auth.currentUser) {
                firebaseUser = auth.currentUser;
                return firebaseUser;
            }
            try {
                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                    const cred = await signInWithCustomToken(auth, __initial_auth_token);
                    firebaseUser = cred.user;
                } else {
                    const cred = await signInAnonymously(auth);
                    firebaseUser = cred.user;
                }
            } catch (err) {
                console.warn('Silent Firebase auth fallback:', err);
                try {
                    const cred = await signInAnonymously(auth);
                    firebaseUser = cred.user;
                } catch (e2) {
                    console.error('Firebase auth unavailable, using local vault mode:', e2);
                }
            }
            return firebaseUser;
        }

        async function loadOwnerConfigFromCloudOrCache() {
            await ensureFirebaseAuth();
            if (firebaseUser) {
                try {
                    const snap = await getDoc(ownerConfigDocRef);
                    if (snap.exists()) {
                        ownerConfig = snap.data();
                        localStorage.setItem(LOCAL_OWNER_CACHE_KEY, JSON.stringify(ownerConfig));
                        return ownerConfig;
                    }
                } catch (err) {
                    console.warn('Could not fetch owner config from Firestore, checking local cache:', err);
                }
            }
            const cached = localStorage.getItem(LOCAL_OWNER_CACHE_KEY);
            if (cached) {
                ownerConfig = JSON.parse(cached);
            }
            return ownerConfig;
        }

        async function saveOwnerConfig(newConfig) {
            ownerConfig = newConfig;
            localStorage.setItem(LOCAL_OWNER_CACHE_KEY, JSON.stringify(ownerConfig));
            await ensureFirebaseAuth();
            if (firebaseUser) {
                try {
                    await setDoc(ownerConfigDocRef, ownerConfig);
                } catch (err) {
                    console.warn('Saved owner config locally; cloud sync queued:', err);
                }
            }
        }

        function isIdentifierAuthorized(identifier) {
            if (!ownerConfig || !identifier) return false;
            const idLower = identifier.trim().toLowerCase();
            const ownerId = (ownerConfig.ownerUsername || ownerConfig.ownerEmail || '').toLowerCase();
            if (idLower === ownerId) return true;
            const extras = ownerConfig.authorizedUsers || [];
            return extras.some(u => (u.identifier || '').toLowerCase() === idLower);
        }

        window.handleAuthSubmit = async function(e) {
            e.preventDefault();
            const identifier = document.getElementById('authEmail').value.trim();
            const identifierLower = identifier.toLowerCase();
            const password = document.getElementById('authPassword').value;
            const errBox = document.getElementById('authErrorBox');
            const submitBtn = document.getElementById('authSubmitBtn');

            errBox.classList.add('hidden');
            submitBtn.disabled = true;
            submitBtn.classList.add('opacity-60');

            try {
                await loadOwnerConfigFromCloudOrCache();
                const passHash = await hashPassword(password);
                const hasExistingOwner = Boolean(ownerConfig && (ownerConfig.ownerUsername || ownerConfig.ownerEmail));

                if (authMode === 'signup') {
                    if (hasExistingOwner) {
                        throw new Error(`This vault is already locked by an owner (${ownerConfig.ownerUsername || ownerConfig.ownerEmail}). Switch to "Owner Sign In" to log in with your password.`);
                    }
                    // Claim ownership with custom username/email + SHA-256 password hash
                    const newOwnerCfg = {
                        ownerUsername: identifierLower,
                        ownerEmail: identifierLower.includes('@') ? identifierLower : '',
                        ownerPasswordHash: passHash,
                        authorizedUsers: [],
                        createdAt: new Date().toISOString()
                    };
                    await saveOwnerConfig(newOwnerCfg);
                    unlockWorkspaceForUser({ identifier: identifierLower, method: 'password' });
                    showToast('Vault created & locked to your account!');
                } else {
                    // Sign In Mode
                    if (!hasExistingOwner) {
                        throw new Error('No owner account exists yet. Click "Create Vault Login" above to set your Username/Email and Password first.');
                    }

                    const ownerId = (ownerConfig.ownerUsername || ownerConfig.ownerEmail || '').toLowerCase();
                    if (identifierLower === ownerId) {
                        if (!ownerConfig.ownerPasswordHash) {
                            throw new Error('This vault was set up with Google Sign-In. Click "Continue with Google Account" below.');
                        }
                        if (passHash !== ownerConfig.ownerPasswordHash) {
                            throw new Error('Invalid password for owner account. Please try again.');
                        }
                        unlockWorkspaceForUser({ identifier: identifierLower, method: 'password' });
                        return;
                    }

                    // Check additional authorized users
                    const matchedUser = (ownerConfig.authorizedUsers || []).find(
                        u => (u.identifier || '').toLowerCase() === identifierLower
                    );
                    if (!matchedUser) {
                        throw new Error('Access Denied: This username/email is not authorized to open this vault.');
                    }
                    if (matchedUser.passwordHash && matchedUser.passwordHash !== passHash) {
                        throw new Error('Invalid password. Please check your credentials.');
                    }
                    unlockWorkspaceForUser({ identifier: identifierLower, method: 'password' });
                }
            } catch (err) {
                errBox.textContent = err.message || 'Authentication failed.';
                errBox.classList.remove('hidden');
            } finally {
                submitBtn.disabled = false;
                submitBtn.classList.remove('opacity-60');
            }
        };

            if (unsubscribeWorkbook) unsubscribeWorkbook();

            if (!firebaseUser) {
                updateSyncBadge('Saved Locally', false);
                return;
            }

            unsubscribeWorkbook = onSnapshot(workbookDocRef, async (docSnap) => {
                if (isLocalSavePending) return;

                if (!docSnap.exists()) {
                    await setDoc(workbookDocRef, { payload: JSON.stringify(workbook), updatedAt: new Date().toISOString() });
                } else {
                    try {
                        const data = docSnap.data();
                        if (data && data.payload) {
                            workbook = JSON.parse(data.payload);
                            localStorage.setItem(LOCAL_WORKBOOK_CACHE_KEY, data.payload);
                        }
                    } catch (err) {

                    }
                }
                renderAllSpreadsheetUI();
                updateSyncBadge('Saved to Cloud', false);
            }, (error) => {
                console.warn('Firestore sync fallback to local cache:', error);
                updateSyncBadge('Saved Locally', false);
            });
        }

        function scheduleCloudSave() {
            isLocalSavePending = true;
            updateSyncBadge('Saving...', false);
            clearTimeout(saveTimeout);

            // Immediately persist to localStorage so zero edits are ever lost
            const serialized = JSON.stringify(workbook);
            localStorage.setItem(LOCAL_WORKBOOK_CACHE_KEY, serialized);

            saveTimeout = setTimeout(async () => {
                try {
                    if (!currentUser) return;
                    await ensureFirebaseAuth();
                    if (firebaseUser) {
                        await setDoc(workbookDocRef, {
                            payload: serialized,
                            updatedAt: new Date().toISOString(),
                            updatedBy: currentUser.identifier
                        });
                        updateSyncBadge('Saved to Cloud', false);
                    } else {
                        updateSyncBadge('Saved Locally', false);
                    }
                } catch (err) {
                    console.warn('Cloud save delayed, preserved locally:', err);
                    updateSyncBadge('Saved Locally', false);
                } finally {
                    isLocalSavePending = false;
                }
            }, 450);
        }

        function colIndexToLetters(idx) {
            let s = '';
            let n = idx + 1;
            while (n > 0) {
                const rem = (n - 1) % 26;
                s = String.fromCharCode(65 + rem) + s;
                n = Math.floor((n - 1) / 26);
            }
            return s;
        }

        function lettersToColIndex(letters) {
            let col = 0;
            const up = letters.toUpperCase();
            for (let i = 0; i < up.length; i++) {
                col = col * 26 + (up.charCodeAt(i) - 64);
            }
            return col - 1;
        }

        function cellIdFromRC(r, c) {
            return `${colIndexToLetters(c)}${r + 1}`;
        }

        function rcFromCellId(cellId) {
            const match = /^([A-Za-z]+)([0-9]+)$/.exec(cellId.trim());
            if (!match) return null;
            return {
                c: lettersToColIndex(match[1]),
                r: parseInt(match[2], 10) - 1
            };
        }

        function expandRangeIds(startId, endId) {
            const s = rcFromCellId(startId);
            const e = rcFromCellId(endId);
            if (!s || !e) return [];
            const rMin = Math.min(s.r, e.r);
            const rMax = Math.max(s.r, e.r);
            const cMin = Math.min(s.c, e.c);
            const cMax = Math.max(s.c, e.c);
            const ids = [];
            for (let r = rMin; r <= rMax; r++) {
                for (let c = cMin; c <= cMax; c++) {
                    ids.push(cellIdFromRC(r, c));
                }
            }
            return ids;
        }

        function evaluateCell(sheet, cellId, visited = new Set()) {
            const cell = sheet.cells[cellId];
            if (!cell || cell.raw === undefined || cell.raw === null || cell.raw === '') return '';
            const rawStr = String(cell.raw);

            if (!rawStr.startsWith('=')) {
                return rawStr;
            }

            if (visited.has(cellId)) return '#REF!';
            const nextVisited = new Set(visited);
            nextVisited.add(cellId);

            try {
                let expr = rawStr.slice(1).trim();

                // Helper to get numeric values from arguments (ranges or individual cells)
                const extractNumbers = (argsStr) => {
                    const nums = [];
                    const parts = splitFormulaArgs(argsStr);
                    parts.forEach(part => {
                        const trimmed = part.trim();
                        if (trimmed.includes(':')) {
                            const [a, b] = trimmed.split(':');
                            expandRangeIds(a, b).forEach(id => {
                                const val = evaluateCell(sheet, id, nextVisited);
                                if (val !== '' && !isNaN(Number(val))) nums.push(Number(val));
                            });
                        } else if (/^[A-Za-z]+[0-9]+$/.test(trimmed)) {
                            const val = evaluateCell(sheet, trimmed.toUpperCase(), nextVisited);
                            if (val !== '' && !isNaN(Number(val))) nums.push(Number(val));
                        } else {
                            const val = evalSubExpression(sheet, trimmed, nextVisited);
                            if (val !== '' && !isNaN(Number(val))) nums.push(Number(val));
                        }
                    });
                    return nums;
                };

                const extractAllValues = (argsStr) => {
                    const vals = [];
                    const parts = splitFormulaArgs(argsStr);
                    parts.forEach(part => {
                        const trimmed = part.trim();
                        if (trimmed.includes(':')) {
                            const [a, b] = trimmed.split(':');
                            expandRangeIds(a, b).forEach(id => {
                                const val = evaluateCell(sheet, id, nextVisited);
                                if (val !== '') vals.push(val);
                            });
                        } else if (/^[A-Za-z]+[0-9]+$/.test(trimmed)) {
                            const val = evaluateCell(sheet, trimmed.toUpperCase(), nextVisited);
                            if (val !== '') vals.push(val);
                        } else {
                            vals.push(evalSubExpression(sheet, trimmed, nextVisited));
                        }
                    });
                    return vals;
                };

                // Process Excel functions from innermost outward
                let safety = 0;
                while (/\b(SUM|AVERAGE|AVG|MIN|MAX|COUNT|COUNTA|ROUND|ABS|IF|CONCAT|LEN)\s*\(/i.test(expr) && safety < 25) {
                    safety++;
                    expr = expr.replace(/\b(SUM|AVERAGE|AVG|MIN|MAX|COUNT|COUNTA|ROUND|ABS|IF|CONCAT|LEN)\s*\(([^()]*)\)/gi, (match, fnName, args) => {
                        const fn = fnName.toUpperCase();
                        if (fn === 'SUM') {
                            const nums = extractNumbers(args);
                            return nums.reduce((a, b) => a + b, 0);
                        }
                        if (fn === 'AVERAGE' || fn === 'AVG') {
                            const nums = extractNumbers(args);
                            return nums.length ? (nums.reduce((a, b) => a + b, 0) / nums.length) : 0;
                        }
                        if (fn === 'MIN') {
                            const nums = extractNumbers(args);
                            return nums.length ? Math.min(...nums) : 0;
                        }
                        if (fn === 'MAX') {
                            const nums = extractNumbers(args);
                            return nums.length ? Math.max(...nums) : 0;
                        }
                        if (fn === 'COUNT') {
                            return extractNumbers(args).length;
                        }
                        if (fn === 'COUNTA') {
                            return extractAllValues(args).length;
                        }
                        if (fn === 'ABS') {
                            const nums = extractNumbers(args);
                            return nums.length ? Math.abs(nums[0]) : 0;
                        }
                        if (fn === 'ROUND') {
                            const parts = splitFormulaArgs(args);
                            const num = Number(evalSubExpression(sheet, parts[0] || '0', nextVisited)) || 0;
                            const digits = Number(evalSubExpression(sheet, parts[1] || '0', nextVisited)) || 0;
                            const factor = Math.pow(10, digits);
                            return Math.round(num * factor) / factor;
                        }
                        if (fn === 'LEN') {
                            const parts = splitFormulaArgs(args);
                            const val = String(evalSubExpression(sheet, parts[0] || '', nextVisited));
                            return val.length;
                        }
                        if (fn === 'CONCAT') {
                            const vals = extractAllValues(args);
                            return `"${vals.join('').replace(/"/g, '\\"')}"`;
                        }
                        if (fn === 'IF') {
                            const parts = splitFormulaArgs(args);
                            const cond = evalCondition(sheet, parts[0] || 'false', nextVisited);
                            const chosen = cond ? (parts[1] || '""') : (parts[2] || '""');
                            const res = evalSubExpression(sheet, chosen, nextVisited);
                            return typeof res === 'string' ? `"${res.replace(/"/g, '\\"')}"` : res;
                        }
                        return 0;
                    });
                }

                return evalSubExpression(sheet, expr, nextVisited);
            } catch (e) {
                return '#ERROR!';
            }
        }

        function splitFormulaArgs(str) {
            const result = [];
            let current = '';
            let inQuotes = false;
            let depth = 0;
            for (let i = 0; i < str.length; i++) {
                const ch = str[i];
                if (ch === '"') inQuotes = !inQuotes;
                if (!inQuotes) {
                    if (ch === '(') depth++;
                    if (ch === ')') depth--;
                    if (ch === ',' && depth === 0) {
                        result.push(current);
                        current = '';
                        continue;
                    }
                }
                current += ch;
            }
            if (current.trim() !== '') result.push(current);
            return result;
        }

        function evalCondition(sheet, condStr, visited) {
            const ops = ['>=', '<=', '<>', '!=', '=', '>', '<'];
            for (const op of ops) {
                const idx = condStr.indexOf(op);
                if (idx !== -1) {
                    const left = evalSubExpression(sheet, condStr.slice(0, idx), visited);
                    const right = evalSubExpression(sheet, condStr.slice(idx + op.length), visited);
                    switch (op) {
                        case '>=': return Number(left) >= Number(right);
                        case '<=': return Number(left) <= Number(right);
                        case '<>':
                        case '!=': return String(left) !== String(right);
                        case '=': return String(left) === String(right);
                        case '>': return Number(left) > Number(right);
                        case '<': return Number(left) < Number(right);
                    }
                }
            }
            const val = evalSubExpression(sheet, condStr, visited);
            return Boolean(val) && val !== '0' && val !== 'FALSE';
        }

        function evalSubExpression(sheet, expr, visited) {
            let trimmed = String(expr).trim();
            if (trimmed === '') return '';
            if ((trimmed.startsWith('"') && trimmed.endsWith('"')) || (trimmed.startsWith("'") && trimmed.endsWith("'"))) {
                return trimmed.slice(1, -1);
            }

            // Replace cell references outside quotes with their evaluated values
            let outsideQuotes = true;
            let rebuilt = '';
            let token = '';

            for (let i = 0; i <= trimmed.length; i++) {
                const ch = trimmed[i] || '';
                if (ch === '"') {
                    if (token) {
                        rebuilt += resolveToken(sheet, token, visited);
                        token = '';
                    }
                    outsideQuotes = !outsideQuotes;
                    rebuilt += ch;
                    continue;
                }
                if (outsideQuotes && /[A-Za-z0-9_]/.test(ch)) {
                    token += ch;
                } else {
                    if (token) {
                        rebuilt += resolveToken(sheet, token, visited);
                        token = '';
                    }
                    rebuilt += ch;
                }
            }

            // If pure string literal remains
            if (/^"[^"]*"$/.test(rebuilt.trim())) {
                return rebuilt.trim().slice(1, -1);
            }

            // Safe arithmetic evaluation
            if (/^[0-9+\-*/().\s%]+$/.test(rebuilt)) {
                const val = Function(`"use strict"; return (${rebuilt});`)();
                if (typeof val === 'number' && !isNaN(val)) {
                    return Math.round(val * 1000000) / 1000000;
                }
            }
            return rebuilt;
        }

        function resolveToken(sheet, token, visited) {
            if (/^[A-Za-z]+[0-9]+$/.test(token)) {
                const val = evaluateCell(sheet, token.toUpperCase(), visited);
                if (val === '') return '0';
                if (!isNaN(Number(val))) return String(Number(val));
                return `"${String(val).replace(/"/g, '\\"')}"`;
            }
            return token;
        }

        function formatDisplayValue(rawEvaluated, style = {}) {
            if (rawEvaluated === '' || rawEvaluated === null || rawEvaluated === undefined) return '';
            const fmt = style.numFormat || 'general';
            const num = Number(rawEvaluated);

            if (!isNaN(num) && rawEvaluated !== '') {
                if (fmt === 'number') {
                    return num.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 });
                }
                if (fmt === 'currency_inr') {
                    return '₹' + num.toLocaleString('en-IN', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
                }
                if (fmt === 'currency_usd') {
                    return '$' + num.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
                }
                if (fmt === 'percent') {
                    return (num * 100).toFixed(1) + '%';
                }
            }
            if (fmt === 'date' && rawEvaluated) {
                const d = new Date(rawEvaluated);
                if (!isNaN(d.getTime())) return d.toLocaleDateString();
            }
            return String(rawEvaluated);
        }

            // Render Column Headers (A, B, C...)
            let headHtml = '<tr><th class="corner-header"></th>';

                    </th>
                `;

            freezeBtn.classList.toggle('bg-sky-100', Boolean(sheet.freezeRow1));
            freezeBtn.classList.toggle('text-sky-800', Boolean(sheet.freezeRow1));

            // Filter indicator
            const hasFilter = activeColumnFilter && activeColumnFilter.sheetId === sheet.id;
            document.getElementById('btnClearFilter').classList.toggle('hidden', !hasFilter);

            // Render Rows
            let bodyHtml = '';
            for (let r = 0; r < sheet.rowCount; r++) {
                // Apply active column filter (never hide row 0 if freezeRow1 is active)
                if (hasFilter && !(sheet.freezeRow1 && r === 0)) {
                    const filterCellId = cellIdFromRC(r, activeColumnFilter.col);
                    const val = String(evaluateCell(sheet, filterCellId)).toLowerCase();
                    if (!val.includes(activeColumnFilter.query.toLowerCase())) {
                        continue;
                    }
                }

                const height = sheet.rowHeights?.[r] || 26;
                const isRowActive = r >= selRMin && r <= selRMax;
                bodyHtml += `<tr style="height:${height}px;">`;
                bodyHtml += `
                    <td class="row-header ${isRowActive ? 'active-row' : ''}" style="height:${height}px;"
                        onclick="selectEntireRow(${r})" oncontextmenu="openRowContextMenu(event, ${r})">
                        ${r + 1}
                        <div class="row-resizer" onmousedown="startRowResize(event, ${r})"></div>
                    </td>
                `;

                for (let c = 0; c < sheet.colCount; c++) {
                    const cellId = cellIdFromRC(r, c);
                    const cellObj = sheet.cells[cellId] || { raw: '', style: {} };
                    const st = cellObj.style || {};
                    const evaluated = evaluateCell(sheet, cellId);
                    const displayVal = formatDisplayValue(evaluated, st);

                    const isMainActive = r === activeCell.r && c === activeCell.c;
                    const isInSel = r >= selRMin && r <= selRMax && c >= selCMin && c <= selCMax;
                    const isFrozen = sheet.freezeRow1 && r === 0;

                    const textDeco = [st.underline ? 'underline' : '', st.strike ? 'line-through' : ''].filter(Boolean).join(' ');
                    const inlineStyles = [
                        st.bgColor ? `background-color:${st.bgColor};` : (isFrozen ? 'background-color:#f8fafc;' : ''),
                        st.color ? `color:${st.color};` : '',
                        st.bold ? 'font-weight:700;' : '',
                        st.italic ? 'font-style:italic;' : '',
                        textDeco ? `text-decoration:${textDeco};` : '',
                        st.align ? `text-align:${st.align};` : (!isNaN(Number(evaluated)) && evaluated !== '' ? 'text-align:right;' : 'text-align:left;'),
                        st.wrap ? 'white-space:normal; word-break:break-word;' : 'white-space:nowrap; overflow:hidden; text-overflow:ellipsis;'
                    ].join('');

                    bodyHtml += `
                        <td id="cell-${cellId}" data-r="${r}" data-c="${c}"
                            class="sheet-cell ${isMainActive ? 'active-cell' : ''} ${isInSel && !isMainActive ? 'in-selection' : ''} ${isFrozen ? 'frozen-row-cell' : ''}"
                            style="${inlineStyles}"
                            onmousedown="handleCellMouseDown(event, ${r}, ${c})"
                            onmouseenter="handleCellMouseEnter(event, ${r}, ${c})"
                            ondblclick="openInlineEditor(${r}, ${c})">
                            ${escapeHtml(displayVal)}
                        </td>
                    `;
                }
                bodyHtml += '</tr>';
            }
            tbody.innerHTML = bodyHtml;
        }

        function updateStatusBarAggregations() {
            const sheet = getActiveSheet();
            const rMin = Math.min(selectionRange.r1, selectionRange.r2);
            const rMax = Math.max(selectionRange.r1, selectionRange.r2);
            const cMin = Math.min(selectionRange.c1, selectionRange.c2);
            const cMax = Math.max(selectionRange.c1, selectionRange.c2);

            const nums = [];
            let countAll = 0;

            for (let r = rMin; r <= rMax; r++) {
                for (let c = cMin; c <= cMax; c++) {
                    const val = evaluateCell(sheet, cellIdFromRC(r, c));
                    if (val !== '') {
                        countAll++;
                        if (!isNaN(Number(val))) nums.push(Number(val));
                    }
                }
            }

            const bar = document.getElementById('statusBarAggregations');
            if (countAll <= 1 && nums.length === 0) {
                bar.innerHTML = `<span>Cell: ${cellIdFromRC(activeCell.r, activeCell.c)}</span>`;
                return;
            }

            if (nums.length > 0) {
                const sum = nums.reduce((a, b) => a + b, 0);
                const avg = sum / nums.length;
                const min = Math.min(...nums);
                const max = Math.max(...nums);
                bar.innerHTML = `
                    <span>Avg: <strong class="text-slate-900">${avg.toLocaleString(undefined, { maximumFractionDigits: 2 })}</strong></span>
                    <span>Count: <strong class="text-slate-900">${countAll}</strong></span>
                    <span>Min: <strong class="text-slate-900">${min.toLocaleString()}</strong></span>
                    <span>Max: <strong class="text-slate-900">${max.toLocaleString()}</strong></span>
                    <span class="bg-emerald-100 text-emerald-900 px-2 py-0.5 rounded font-bold">Sum: ${sum.toLocaleString(undefined, { maximumFractionDigits: 2 })}</span>
                `;
            } else {
                bar.innerHTML = `<span>Count: <strong class="text-slate-900">${countAll}</strong></span>`;
            }
        }

            const sheet = getActiveSheet();

                const nr = Math.max(0, Math.min(sheet.rowCount - 1, selectionRange.r2 + dr));
                const nc = Math.max(0, Math.min(sheet.colCount - 1, selectionRange.c2 + dc));
                selectionRange.r2 = nr;
                selectionRange.c2 = nc;
            } else {
                const nr = Math.max(0, Math.min(sheet.rowCount - 1, activeCell.r + dr));
                const nc = Math.max(0, Math.min(sheet.colCount - 1, activeCell.c + dc));
                activeCell = { r: nr, c: nc };
                selectionRange = { r1: nr, c1: nc, r2: nr, c2: nc };

            renderGrid();
            syncToolbarWithActiveCell();

            if (td) td.scrollIntoView({ block: 'nearest', inline: 'nearest' });

        // Global Keyboard Navigation & Shortcuts
        document.addEventListener('keydown', (e) => {
            if (!currentUser || isEditingCell) return;
            const tag = document.activeElement?.tagName;
            if (tag === 'INPUT' || tag === 'SELECT' || tag === 'TEXTAREA') return;

            if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'z') {
                e.preventDefault();
                undoAction();
                return;
            }
            if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'y') {
                e.preventDefault();
                redoAction();
                return;
            }
            if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'b') {
                e.preventDefault();
                toggleCellFormat('bold');
                return;
            }
            if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'i') {
                e.preventDefault();
                toggleCellFormat('italic');
                return;
            }
            if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'f') {
                e.preventDefault();
                openFindReplaceModal();
                return;
            }
            if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'c') {
                e.preventDefault();
                copySelection();
                return;
            }
            if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'v') {
                e.preventDefault();
                pasteSelection();
                return;
            }

            switch (e.key) {
                case 'ArrowUp':
                    e.preventDefault();
                    moveActiveCell(-1, 0, e.shiftKey);
                    break;
                case 'ArrowDown':
                    e.preventDefault();
                    moveActiveCell(1, 0, e.shiftKey);
                    break;
                case 'ArrowLeft':
                    e.preventDefault();
                    moveActiveCell(0, -1, e.shiftKey);
                    break;
                case 'ArrowRight':
                    e.preventDefault();
                    moveActiveCell(0, 1, e.shiftKey);
                    break;
                case 'Tab':
                    e.preventDefault();
                    moveActiveCell(0, e.shiftKey ? -1 : 1);
                    break;
                case 'Enter':
                    e.preventDefault();
                    openInlineEditor(activeCell.r, activeCell.c);
                    break;
                case 'Delete':
                case 'Backspace':
                    e.preventDefault();
                    clearSelectedCells();
                    break;
                default:
                    if (e.key.length === 1 && !e.ctrlKey && !e.metaKey && !e.altKey) {
                        e.preventDefault();
                        openInlineEditor(activeCell.r, activeCell.c, e.key);
                    }
            }
        });

        function copySelection() {
            const sheet = getActiveSheet();
            const rMin = Math.min(selectionRange.r1, selectionRange.r2);
            const rMax = Math.max(selectionRange.r1, selectionRange.r2);
            const cMin = Math.min(selectionRange.c1, selectionRange.c2);
            const cMax = Math.max(selectionRange.c1, selectionRange.c2);

            const grid = [];
            const textRows = [];
            for (let r = rMin; r <= rMax; r++) {
                const rowArr = [];
                const tsvRow = [];
                for (let c = cMin; c <= cMax; c++) {
                    const cellId = cellIdFromRC(r, c);
                    const cell = sheet.cells[cellId] || { raw: '', style: {} };
                    rowArr.push(JSON.parse(JSON.stringify(cell)));
                    tsvRow.push(evaluateCell(sheet, cellId));
                }
                grid.push(rowArr);
                textRows.push(tsvRow.join('\t'));
            }
            clipboardCells = grid;

            // Also copy TSV to system clipboard via execCommand fallback
            const ta = document.createElement('textarea');
            ta.value = textRows.join('\n');
            document.body.appendChild(ta);
            ta.select();
            document.execCommand('copy');
            document.body.removeChild(ta);

            showToast('Copied selection');
        }

        function pasteSelection() {
            if (!clipboardCells) return;
            const sheet = getActiveSheet();
            pushUndoSnapshot();

            for (let dr = 0; dr < clipboardCells.length; dr++) {
                for (let dc = 0; dc < clipboardCells[dr].length; dc++) {
                    const tr = activeCell.r + dr;
                    const tc = activeCell.c + dc;
                    if (tr < sheet.rowCount && tc < sheet.colCount) {
                        sheet.cells[cellIdFromRC(tr, tc)] = JSON.parse(JSON.stringify(clipboardCells[dr][dc]));
                    }
                }
            }
            renderAllSpreadsheetUI();
            scheduleCloudSave();
            showToast('Pasted cells');
        }

        window.sortActiveColumn = function(direction) {
            const sheet = getActiveSheet();
            pushUndoSnapshot();

            const startRow = sheet.freezeRow1 ? 1 : 0;
            const col = activeCell.c;
            const rowsData = [];

            for (let r = startRow; r < sheet.rowCount; r++) {
                const rowCells = [];
                let hasAnyData = false;
                for (let c = 0; c < sheet.colCount; c++) {
                    const cell = sheet.cells[cellIdFromRC(r, c)] || { raw: '', style: {} };
                    if (cell.raw !== '') hasAnyData = true;
                    rowCells.push(cell);
                }
                const sortKey = evaluateCell(sheet, cellIdFromRC(r, col));
                rowsData.push({ rowCells, sortKey, hasAnyData });
            }

            const nonEmptyRows = rowsData.filter(x => x.hasAnyData);
            const emptyRows = rowsData.filter(x => !x.hasAnyData);

            nonEmptyRows.sort((a, b) => {
                const numA = Number(a.sortKey);
                const numB = Number(b.sortKey);
                let cmp = 0;
                if (a.sortKey !== '' && b.sortKey !== '' && !isNaN(numA) && !isNaN(numB)) {
                    cmp = numA - numB;
                } else {
                    cmp = String(a.sortKey).localeCompare(String(b.sortKey), undefined, { numeric: true });
                }
                return direction === 'asc' ? cmp : -cmp;
            });

            const merged = [...nonEmptyRows, ...emptyRows];
            merged.forEach((item, idx) => {
                const targetR = startRow + idx;
                for (let c = 0; c < sheet.colCount; c++) {
                    sheet.cells[cellIdFromRC(targetR, c)] = item.rowCells[c];
                }
            });

            renderAllSpreadsheetUI();
            scheduleCloudSave();
            showToast(`Sorted Column ${colIndexToLetters(col)} (${direction.toUpperCase()})`);
        };

        window.executeFindNext = function() {
            const q = document.getElementById('findQueryInput').value.toLowerCase();
            if (!q) return;
            const sheet = getActiveSheet();

            for (let r = activeCell.r; r < sheet.rowCount; r++) {
                const startC = (r === activeCell.r) ? activeCell.c + 1 : 0;
                for (let c = startC; c < sheet.colCount; c++) {
                    const id = cellIdFromRC(r, c);
                    const val = String(sheet.cells[id]?.raw || '').toLowerCase();
                    if (val.includes(q)) {
                        activeCell = { r, c };
                        selectionRange = { r1: r, c1: c, r2: r, c2: c };
                        renderGrid();
                        syncToolbarWithActiveCell();
                        document.getElementById('findResultSummary').textContent = `Found at ${id}`;
                        return;
                    }
                }
            }
            document.getElementById('findResultSummary').textContent = 'Reached end of sheet.';
        };

        window.executeReplaceAll = function() {
            const q = document.getElementById('findQueryInput').value;
            const rep = document.getElementById('replaceQueryInput').value;
            if (!q) return;
            const sheet = getActiveSheet();
            pushUndoSnapshot();
            let count = 0;

            Object.keys(sheet.cells).forEach(id => {
                const raw = String(sheet.cells[id].raw || '');
                if (raw.includes(q)) {
                    sheet.cells[id].raw = raw.split(q).join(rep);
                    count++;
                }
            });

            renderAllSpreadsheetUI();
            scheduleCloudSave();
            document.getElementById('findResultSummary').textContent = `Replaced ${count} occurrence(s).`;
        };

        window.exportActiveSheetCSV = function() {
            const sheet = getActiveSheet();
            const lines = [];
            for (let r = 0; r < sheet.rowCount; r++) {
                const rowVals = [];
                let hasData = false;
                for (let c = 0; c < sheet.colCount; c++) {
                    const val = String(evaluateCell(sheet, cellIdFromRC(r, c)));
                    if (val !== '') hasData = true;
                    rowVals.push(`"${val.replace(/"/g, '""')}"`);
                }
                if (hasData) lines.push(rowVals.join(','));
            }
            const blob = new Blob(['\uFEFF' + lines.join('\n')], { type: 'text/csv;charset=utf-8;' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `${sheet.name.replace(/\s+/g, '_')}.csv`;
            a.click();
            URL.revokeObjectURL(url);
            showToast(`Exported ${sheet.name}.csv`);
        };

        window.importCSVFile = function(event) {
            const file = event.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = function(e) {
                const text = e.target.result;
                const rows = parseCSVText(text);
                const newSheet = {
                    id: 'sheet-' + Date.now(),
                    name: file.name.replace(/\.csv$/i, '').slice(0, 24) || 'Imported CSV',
                    rowCount: Math.max(40, rows.length + 10),
                    colCount: Math.max(16, (rows[0]?.length || 0) + 4),
                    freezeRow1: true,
                    colWidths: {},
                    rowHeights: {},
                    cells: {}
                };
                rows.forEach((rowArr, r) => {
                    rowArr.forEach((val, c) => {
                        newSheet.cells[cellIdFromRC(r, c)] = {
                            raw: val,
                            style: r === 0 ? { bold: true, bgColor: '#dcfce7' } : {}
                        };
                    });
                });
                workbook.sheets.push(newSheet);
                switchSheet(newSheet.id);
                showToast(`Imported ${rows.length} rows into "${newSheet.name}"`);
                event.target.value = '';
            };
            reader.readAsText(file);
        };

        function parseCSVText(text) {
            const rows = [];
            let row = [];
            let current = '';
            let inQuotes = false;
            for (let i = 0; i < text.length; i++) {
                const ch = text[i];
                const next = text[i + 1];
                if (ch === '"' && inQuotes && next === '"') {
                    current += '"';
                    i++;
                } else if (ch === '"') {
                    inQuotes = !inQuotes;
                } else if (ch === ',' && !inQuotes) {
                    row.push(current);
                    current = '';
                } else if ((ch === '\n' || (ch === '\r' && next === '\n')) && !inQuotes) {
                    if (ch === '\r') i++;
                    row.push(current);
                    rows.push(row);
                    row = [];
                    current = '';
                } else {
                    current += ch;
                }
            }
            if (current !== '' || row.length > 0) {
                row.push(current);
                rows.push(row);
            }
            return rows;
        }

        function escapeHtml(str) {
            return String(str ?? '')
                .replace(/&/g, '&amp;')
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;')
                .replace(/"/g, '&quot;')
                .replace(/'/g, '&#039;');
        }
